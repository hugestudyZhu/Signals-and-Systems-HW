%%1.5b
n = 0:30;
x1 = double(n==0);
x2 = ones(1,length(n));
 
y1 = diffeqn(1,x1,0);
y2 = diffeqn(1,x2,0);
 
figure;
subplot(2,1,1);
stem(n,y1,'filled');
title('(b) x_1[n]=δ[n], y[-1]=0');
xlabel('n'); ylabel('y_1[n]');
grid on;
 
subplot(2,1,2);
stem(n,y2,'filled');
title('(b) x_2[n]=u[n], y[-1]=0');
xlabel('n'); ylabel('y_2[n]');
grid on;
 
%%1.5c
n = 0:30;
x1 = ones(1,length(n));
x2 = 2*ones(1,length(n));
 
y1c = diffeqn(1,x1,-1);
y2c = diffeqn(1,x2,-1);
d = 2*y1c - y2c;
 
figure;
subplot(3,1,1);
stem(n,y1c,'filled');
title('(c) y_1[n], 输入 u[n], y[-1]=-1');
xlabel('n'); ylabel('y_1[n]');
grid on;
 
subplot(3,1,2);
stem(n,y2c,'filled');
title('(c) y_2[n], 输入 2u[n], y[-1]=-1');
xlabel('n'); ylabel('y_2[n]');
grid on;
 
subplot(3,1,3);
stem(n,d,'filled');
title('(c) 2y_1[n] - y_2[n]');
xlabel('n'); ylabel('差值');
grid on;
 
%%1.5a
function y = diffeqn(a,x,yn1)
N = length(x);
y = zeros(1,N);
y(1) = a*yn1 + x(1);
for n = 2:N
y(n) = a*y(n-1) + x(n);
end
end

(4)结果分析
(b)当 a=1, y[-1]=0 时，系统是累加器。输入 δ[n] 得到 y1[n] = u[n]。输入 u[n] 得到 y2[n] = (n+1)u[n]。
(c) 虽然差分方程是线性的，但初始条件 y[-1] = -1 不为零。系统的完全响应 = 零输入响应 + 零状态响应。两个输入的零输入响应相同,所以 2y1 - y2 带有一个零输入响应。
